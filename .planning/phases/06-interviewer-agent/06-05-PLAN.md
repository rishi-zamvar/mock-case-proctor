export function buildTurnRequest(
  s: Pick<SessionFile, 'machineState' | 'transcript' | 'controlLog' | 'openingPlan'>,
  pending: { kind: 'grill' | 'wrapup'; section?: SectionId; hintAllowed?: boolean },
  now: number
): TurnRequest

Rules: directive = wrapup when kind==='wrapup'; else 'framing' if turnsInSection===0; else 'wrap_section' if caps && turnsInSection >= caps.soft; else 'probe'. grillElapsedMs = grillStartedAt ? max(0, now - grillStartedAt) : 0. budgetMs = config.budgetMin*60_000. pushback = pushbackStatus(machineState). lastAnswerLong = last candidate transcript turn's id === machineState.interruptedAnswerId. surfacedFlagIds = unique flags_surfaced across controlLog in order. openingLabels = done.length===0 && turnsInSection===0 && kind==='grill' ?
