// src/shared/ipc.ts
CH.trendsGet = 'trends:get'
export const DIMENSION_IDS = ['clear_call','lead_edge_8','retention','valuation','back_solve','quality_poise'] as const
export type DimensionId = (typeof DIMENSION_IDS)[number]
export const DIMENSION_LABELS: Record<DimensionId, string> = { clear_call: 'Clear call', lead_edge_8: 'Lead Edge 8',
  retention: 'Retention insight', valuation: 'Valuation and comps', back_solve: 'Return back-solve',
  quality_poise: 'Business quality and poise' }
export type Band = 'ready' | 'close' | 'another'          // 26-30 / 20-25 / <20
export type EvidenceQuote = { quote: string; turnId: string }
export type RubricRow = { id: DimensionId; dimension: string; score: number /*1-5*/; evidence: EvidenceQuote[] /*0-3, verified only*/;
  answerKey: string; rationale: string; unverified: boolean }
export type FlagStatus = 'found' | 'hinted' | 'missed'     // unchanged; 'hinted' = found after a hint
export type FlagRowView = { id: string; flag: string; status: FlagStatus; where: string /* deck slide refs, e.g. "S12, A3" */ }
export type ChecksSummary = { numbersCorrect: number; numbersWrong: number; numbersUnverifiable: number;
  brevity: Record<BrevityBand, number>; avgAnswerMs: number | null; ledWithCall: boolean;
  flagsFound: number; flagsHinted: number; flagsTotal: number }
export type Scorecard = { rubric: RubricRow[]; total: number; band: Band; flags: FlagRowView[];
  keep: string; fix: string; insight: { calledOut: string[]; missed: string[] }; summary: string;
  evidenceStats: { proposed: number; verified: number; dropped: number };
  checksSummary: ChecksSummary; graderModel: string; promptVersion: string; scoredAt: number }
export type NumberCheck = { claim: string; matched: string | null; expected: string | null; correct: boolean | 'unverifiable' }
export type Checks = { brevity: { turnId: string; ms: number; band: BrevityBand }[]; ledWithCall: boolean; omitted: string[]; numbers: NumberCheck[] }
export type RevealLine = { id: string; speaker: Speaker; text: string; offsetMs: number; ms: number | null; section: string | null }
export type Reveal = { scorecard: Scorecard; checks: Checks; idealAnswer: string; opening: string;
  keyNumbers: { label: string; text: string }[]; intendedRead: string; backSolve: string[]; transcript: RevealLine[] }
ViewState.scoringFailed?: boolean                      // optional in the type; toViewState ALWAYS sets it
Intent |= { type: 'retryScore' }
export type SessionSummary = { sessionId: string; caseId: string; at: number /*scoredAt*/; total: number; band: Band;
  dims: Record<DimensionId, number>; flagFoundRate: number | null; avgAnswerMs: number | null;
  numberAccuracy: number | null; brevity: Record<BrevityBand, number>; fix: string }
export type TrendsInsights = { recurringWeaknesses: string[];
  progress: { id: DimensionId; direction: 'up' | 'flat' | 'down'; note: string }[];
  carriedForwardFix: { fix: string; improved: 'yes' | 'no' | 'unclear'; note: string } | null;
  nextCase: { caseId: string; why: string } | null; readiness: string }
export type TrendsView = { status: 'insufficient' | 'ready' | 'failed'; summaries: SessionSummary[] /*oldest first*/; insights: TrendsInsights | null }
Api.trends?: { get(): Promise<TrendsView> }          // optional: old renderer test mocks omit it
