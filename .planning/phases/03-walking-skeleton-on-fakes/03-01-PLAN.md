// ---- src/shared/ipc.ts additions ----
export const CH = { keyStatus, keySave, keyClear,
  libraryList: 'library:list', sessionStart: 'session:start', sessionIntent: 'session:intent',
  sessionResume: 'session:resume', sessionDiscard: 'session:discard', speakAck: 'speak:ack',
  view: 'view:state', speak: 'speak:sentence', e2eSay: 'e2e:say' } as const

export type ViewPhase = 'warmup'|'review'|'presentation'|'extraction'|'grilling'|'wrapup'|'scoring'|'reviewMode'
export type Speaker = 'interviewer' | 'candidate'
export type TranscriptLine = { id: string; speaker: Speaker; text: string }
export type SectionView = { id: string; label: string; index: number; total: number }
export type RubricRow = { dimension: string; score: number; evidence: string; answerKey: string }
export type FlagStatus = 'found' | 'hinted' | 'missed'
export type FlagRowView = { id: string; flag: string; status: FlagStatus }
export type Scorecard = { rubric: RubricRow[]; total: number; band: 'strong'|'average'|'weak'; flags: FlagRowView[]; graderModel: string; promptVersion: string }
export type BrevityBand = 'short'|'brief'|'ideal'|'long'|'flagged'
export type Checks = { brevity: { turnId: string; ms: number; band: BrevityBand }[]; ledWithCall: boolean; omitted: string[]; numbers: { said: string; expected: string; ok: boolean }[] }
export type Reveal = { scorecard: Scorecard; checks: Checks; idealAnswer: string; opening: string; keyNumbers: { label: string; text: string }[] }
export type ViewState = {
  sessionId: string; phase: ViewPhase; card: CaseCard
  reviewDeadline: number | null   // app-clock epoch ms
  clockNow: number                // app-clock epoch ms at push time
  timeScale: number               // app-clock ms per real ms (1 in production)
  section: SectionView | null
  transcript: TranscriptLine[]    // candidate answers + interviewer lines ACTUALLY spoken
  speaking: boolean; canTalk: boolean; endedEarly: boolean
  reveal: Reveal | null           // non-null ONLY when phase === 'reviewMode'
}
export type LibraryEntry = { card: CaseCard; completed: { at: number; score: number } | null }
export type ResumableSession = { sessionId: string; caseId: string; company: string; phase: ViewPhase }
export type LibraryView = { cases: LibraryEntry[]; resumable: ResumableSession | null; quarantined: number; error: string | null }
export type Intent = { type: 'answer'; ms: number } | { type: 'ready' } | { type: 'skipWarmup' } | { type: 'end' }
export type StartOptions = { warmup: boolean }
export type SpeakEvent = { turnId: string; seq: number; text: string }
export interface Api {
  key: { ...unchanged }
  library: { list(): Promise<LibraryView> }
  session: {
    start(caseId: string, opts: StartOptions): Promise<{ sessionId: string }>
    intent(i: Intent): Promise<void>
    resume(id: string): Promise<{ sessionId: string }>
    discard(id: string): Promise<void>
  }
  speak: { ack(turnId: string, seq: number): Promise<void> }
  onView(cb: (v: ViewState) => void): () => void
  onSpeak(cb: (s: SpeakEvent) => void): () => void
}
export type IpcTapEntry = { channel: string; payload: unknown }
export interface E2EHooks { micPeak(ms: number): Promise<number>; say(text: string): Promise<void>; ipcLog(): IpcTapEntry[] }

// ---- src/main/ports/index.ts (types only, main-only) ----
import type { InterviewControl } from '../domain/machine/interviewMachine'
import type { SectionId } from '../domain/machine/sections'
import type { InterviewerBrief } from '../domain/cases/brief'
import type { Scorecard } from '../../shared/ipc'
import type { Turn } from '../store/sessionSchema'
export type LlmChunk = { type: 'text'; delta: string } | { type: 'control'; payload: InterviewControl } | { type: 'done'; warmupComplete?: boolean }
export type HistoryTurn = { speaker: 'interviewer' | 'candidate'; text: string }
export type TurnRequest = { kind: 'grill' | 'wrapup'; section?: SectionId; hintAllowed?: boolean; turnInSection: number; history: HistoryTurn[] }
export type ScoreInput = { transcript: Turn[]; controlLog: { at: number; turnId: string; control: InterviewControl }[]; endedEarly: boolean; omitted: SectionId[] }
export interface LlmPort {
  warmupTurn(req: { history: HistoryTurn[] }, signal: AbortSignal): AsyncIterable<LlmChunk>
  streamTurn(req: TurnRequest, signal: AbortSignal): AsyncIterable<LlmChunk>
  extractPlan(opening: string, signal: AbortSignal): Promise<{ order: string[] }>
  scoreSession(input: ScoreInput, signal: AbortSignal): Promise<Scorecard>
}
/** One interviewer per case session (INT-08): the brief is bound at construction, never shared across sessions. */
export type MakeLlm = (brief: InterviewerBrief) => LlmPort
export interface SttPort { transcribe(pcm16k: Float32Array): Promise<{ text: string }> }
export interface ClockPort { now(): number; setTimer(at: number, cb: () => void): () => void; readonly scale: number }

// ---- src/main/store/sessionSchema.ts (main-only) ----
export const InterviewStateSchema  // z.object strict mirror of InterviewState (phase enum of the 9 Phase values, config object, SectionId enums from SECTION_IDS, caps nullable)
export const TurnSchema = z.object({ id: z.string().min(1), speaker: z.enum(['interviewer','candidate']), phase: PhaseEnum, section: SectionEnum.optional(), text: z.string(), at: z.number(), ms: z.number().nonnegative().optional() }).strict()
export const PendingTurnSchema = z.object({ kind: z.enum(['warmup','extraction','grill','wrapup','score']), section: SectionEnum.optional(), hintAllowed: z.boolean().optional() }).strict()
export const SessionFileSchema = z.object({
  schemaVersion: z.literal(1), id: z.string().regex(/^[A-Za-z0-9_-]{1,80}$/), caseId: z.string().regex(/^\d{2}$/),
  startedAt: z.number(), endedAt: z.number().optional(), status: z.enum(['active','completed','abandoned']), warmup: z.boolean(),
  machineState: InterviewStateSchema, transcript: z.array(TurnSchema),
  controlLog: z.array(z.object({ at: z.number(), turnId: z.string(), control: InterviewControlSchema }).strict()),
  timings: z.array(z.object({ turnId: z.string(), kind: z.string(), startedAt: z.number(), firstTextAt: z.number().optional(), doneAt: z.number() }).strict()),
  pending: PendingTurnSchema.nullable(), scorecard: ScorecardSchema.optional() }).strict()
export type SessionFile = z.infer<typeof SessionFileSchema>; export type Turn = z.infer<typeof TurnSchema>; export type PendingTurn = z.infer<typeof PendingTurnSchema>
