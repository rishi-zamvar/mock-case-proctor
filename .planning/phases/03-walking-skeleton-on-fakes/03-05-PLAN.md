Rules:
- phase: s.machineState.phase, with 'idle' mapped to 'warmup'
- transcript: s.transcript.map(t => ({ id: t.id, speaker: t.speaker, text: t.text }))
- section: grilling only: current → { id, label from SECTION_CATALOG, index: done.length, total: plan.length }, else null
- canTalk: !speaking && phase in ['warmup','presentation','grilling']
- endedEarly: s.machineState.endedEarly; reviewDeadline: s.machineState.reviewDeadline
- reveal: phase === 'reviewMode' && s.scorecard ? buildReveal(s, brief) : null
- buildReveal: { scorecard: s.scorecard, checks, idealAnswer: brief.idealAnswer, opening: first candidate turn with phase 'presentation' (text) or '', keyNumbers: brief.memorize }
- checks: brevity = candidate turns with ms !== undefined in phases presentation|grilling → {turnId, ms, band: brevityBand(ms)}; ledWithCall = /\b(invest|pass|buy|my call)\b/i on the opening's first sentence; omitted = machineState.omitted mapped to SECTION_CATALOG labels; numbers = []
