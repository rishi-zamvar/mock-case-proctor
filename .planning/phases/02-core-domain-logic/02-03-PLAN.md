```typescript
// src/main/domain/machine/sections.ts
export const SECTION_IDS = ['business','lead_edge_8','retention','industry','diligence','valuation','call'] as const
export type SectionId = (typeof SECTION_IDS)[number]
export type RubricDimension = 'Clear call' | 'Lead Edge 8' | 'Retention insight' | 'Valuation and comps' | 'Return back-solve' | 'Business quality and poise'
export const SECTION_CATALOG: readonly { id: SectionId; label: string; rubric: readonly RubricDimension[] }[]
  // business→['Business quality and poise'], lead_edge_8→['Lead Edge 8'], retention→['Retention insight'],
  // industry→['Valuation and comps'] (qual+quant incl. comps), diligence→['Business quality and poise'],
  // valuation→['Return back-solve','Valuation and comps'] (work back to 3x), call→['Clear call']
export function mergePlan(order: readonly string[]): { plan: SectionId[]; omitted: SectionId[] }
export function turnCaps(budgetMin: number, sectionCount: number): { soft: number; hard: number }

// src/main/domain/machine/interviewMachine.ts
export type Phase = 'idle'|'warmup'|'review'|'presentation'|'extraction'|'grilling'|'wrapup'|'scoring'|'reviewMode'
export type MachineConfig = { warmup: boolean; warmupPosition: 'before'|'after'; budgetMin: number; reviewMs: number }
export const DEFAULT_CONFIG: MachineConfig  // { warmup: true, warmupPosition: 'before', budgetMin: 25, reviewMs: 45*60_000 }
export type InterviewControl = {           // minimal; Phase 6 adds the zod schema and must conform
  section: SectionId; section_status: 'continue'|'sufficient'; next_section?: SectionId
  probe_failed?: boolean; hint_given?: boolean; flags_surfaced?: string[]
}
export type InterviewEvent =               // EVERY event carries now (ms epoch) — timers as data
  | { type: 'START'; now: number; config?: Partial<MachineConfig> }
  | { type: 'WARMUP_DONE'; now: number } | { type: 'WARMUP_SKIPPED'; now: number }
  | { type: 'READY'; now: number } | { type: 'REVIEW_DEADLINE'; now: number }
  | { type: 'PRESENTATION_DONE'; now: number }
  | { type: 'SECTION_PLAN_SET'; now: number; order: string[] }
  | { type: 'ANSWER_TICK'; now: number; answerId: string; elapsedMs: number }
  | { type: 'ANSWER_FINAL'; now: number; answerId: string; text: string; ms: number }
  | { type: 'AGENT_TURN_DONE'; now: number; control: InterviewControl | null }
  | { type: 'WRAPUP_DONE'; now: number } | { type: 'END_REQUESTED'; now: number } | { type: 'SCORED'; now: number }
export type Effect =
  | { type: 'PHASE_CHANGED'; from: Phase; to: Phase; at: number }
  | { type: 'START_TIMER'; id: 'review'; deadline: number }
  | { type: 'REQUEST_LLM_TURN'; kind: 'extraction'|'grill'|'wrapup'|'score'; section?: SectionId; hintAllowed?: boolean }
  | { type: 'INTERRUPT_SUGGESTED'; answerId: string }
  | { type: 'PERSIST' }
export type InterviewState = {
  phase: Phase; config: MachineConfig; startedAt: number | null; reviewDeadline: number | null
  plan: SectionId[]; omitted: SectionId[]; done: SectionId[]; current: SectionId | null
  caps: { soft: number; hard: number } | null; turnsInSection: number; failedProbes: number; hintAllowed: boolean
  interruptedAnswerId: string | null; endedEarly: boolean
}
export function initialState(): InterviewState
export function reduce(state: InterviewState, event: InterviewEvent): { state: InterviewState; effects: Effect[] }
```
