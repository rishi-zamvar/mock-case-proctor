function fnRef(k) {
  const alpha = T.house(D.bank.house).markers === 'alpha';   // H3 = alpha, H1/H2 = numeric
  const lists = [...Object.entries(D.deck).map(([t, sl]) => [t, sl && sl.notes]),
                 ...(D.inserts || []).map((x) => [x.id.toLowerCase(), x.notes])];
  const [t, id] = k.includes('.') ? k.split('.') : [null, k];
  const hits = lists.filter(([tt, n]) => (!t || tt === t) && n && n.numbered)
                    .map(([, n]) => n.numbered.findIndex((x) => x.id === id)).filter((i) => i >= 0);
  if (hits.length !== 1) throw new Error(`{fn:${k}} matches ${hits.length} notes; ...`);
  return `note (${alpha ? String.fromCharCode(97 + hits[0]) : hits[0] + 1})`;
}
// R(s): s.replace(/\{fn:([\w.]+)\}/g, fnRef).replace(/\{(\w+)\}/g, k => tokens[k].s or throw)
