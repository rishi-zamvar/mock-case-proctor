export interface InterviewerBrief extends LeakSource {   // leakSources + deckText satisfy checkUtterance(text, brief)
  caseId: string; level: number; company: { name: string; oneLiner: string; tagline: string }; vertical: string
  intendedRead: { short: string; bucket: string; sentence: string; answerKey: unknown }
  business: { plain: string; facts: { label: string; text: string }[] }; glossary: { term: string; def: string }[]
  deckFacts: { deckNumbers: unknown[]; snapshot: unknown[] }; deckPages: string[]
  leadEdge8: { rows: unknown[]; weakest: unknown; weakestWhy: unknown; scores: unknown; lines: unknown }
  flags: { hidden: unknown[]; map: { id: string; flag: string; kind: string; role: string; templates: string[]; anchor: string; notice: string; say: string }[]; deliberateOmissions: string; bankerRationale: string | null; bankerRationaleWhy: string | null }
  mustPullOuts: unknown[]; whatItTests: { skill: string; evidence: string }[]
  comps: { iv: unknown; pb: unknown[]; rows: unknown[] }; precedents: unknown[]; kpis: unknown[]
  tailwinds: unknown[]; headwinds: unknown[]; exitBuyers: unknown[]
  backSolve: unknown; priceLadder: { acceptedCalls: unknown[]; condition: unknown; header: unknown }
  pushback: { questions: unknown[]; mustAsk: unknown[]; convinceMe: unknown; convinceMeStrong: unknown; doNotUseUnlessRaised: unknown[]; genericBank: unknown[] }
  rubric: { dimension: string; five: string; one: string }[]; rubricExtras: unknown
  idealAnswer: string; modelAnswer: Record<string, string>
  memorize: { label: string; text: string }[]; traps: string[]; annotations: unknown[]
}
export function buildBrief(data: unknown, deckText: string): InterviewerBrief

export const FINANCE_VOCAB: readonly string[]  // ARR, NRR, GRR, CAGR, EV, NTM, pre-money, post-money, MOIC, Rule of 40, burn multiple
export function sttPrompt(brief: Pick<InterviewerBrief, 'company' | 'glossary'>): string
