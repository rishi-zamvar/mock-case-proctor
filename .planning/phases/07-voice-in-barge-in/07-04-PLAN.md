```ts
// src/main/stt/protocol.ts (types only)
export type ToWorker = { type: 'init'; model: string } | { type: 'transcribe'; id: number; pcm: Float32Array; prompt?: string }
export type FromWorker =
  | { type: 'ready'; gpus: { name: string; description: string; type: string }[] }
  | { type: 'text'; id: number; text: string }
  | { type: 'error'; id?: number; code: 'init_failed' | 'transcribe_failed' }

// src/main/stt/whisperStt.ts (Electron-free, no native import)
export const MODEL_FILES = { turbo: 'ggml-large-v3-turbo-q5_0.bin', base: 'ggml-base.en.bin' } as const
export type SttModel = keyof typeof MODEL_FILES
export type WhisperStatus = 'metal' | 'cpu' | 'failed'
export type ChildLike = {
  postMessage(m: ToWorker): void
  on(ev: 'message', cb: (m: FromWorker) => void): unknown
  on(ev: 'exit', cb: (code: number) => void): unknown
  kill(): boolean
}
export class SttCrashed extends Error {}           // name = 'SttCrashed'
export function createWhisperStt(o: {
  fork(): ChildLike
  model: string            // absolute path to the .bin
  log: Logger
  timeoutMs?: number       // per request, default 60_000 -> rejects SttCrashed('timeout') and kills the child
}): SttPort & { warm(): Promise<void>; status(): Promise<WhisperStatus>; dispose(): void }
```
